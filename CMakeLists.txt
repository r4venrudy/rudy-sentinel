cmake_minimum_required(VERSION 3.24)
project(RudySentinel VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(JSON_BuildTests OFF CACHE INTERNAL "")
FetchContent_MakeAvailable(json)
add_library(rs_engine agent/engine.cpp)
target_link_libraries(rs_engine PUBLIC nlohmann_json::nlohmann_json)
target_include_directories(rs_engine PUBLIC agent shared)
add_executable(rudy-sentinel agent/main.cpp agent/collector.cpp)
target_link_libraries(rudy-sentinel PRIVATE rs_engine)
option(RS_EVIDENCE "Enable offline Rudy Evidence bundle export using an installed SDK" OFF)
if(RS_EVIDENCE)
  find_package(RudyEvidence 0.1 CONFIG REQUIRED)
  target_sources(rudy-sentinel PRIVATE agent/evidence.cpp)
  target_link_libraries(rudy-sentinel PRIVATE RudyEvidence::core)
  target_compile_definitions(rudy-sentinel PRIVATE RS_EVIDENCE=1)
endif()
if(WIN32)
  target_link_libraries(rudy-sentinel PRIVATE advapi32 shell32)
endif()
if(MSVC)
  target_compile_options(rs_engine PRIVATE /W4 /WX /utf-8)
  target_compile_options(rudy-sentinel PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(rs_engine PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(rudy-sentinel PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_executable(engine-tests tests/engine_tests.cpp)
target_link_libraries(engine-tests PRIVATE rs_engine)
add_test(NAME engine COMMAND engine-tests ${CMAKE_CURRENT_SOURCE_DIR}/rules/default.json)
add_executable(queue-tests tests/queue_tests.c shared/queue.c)
add_test(NAME queue COMMAND queue-tests)
add_test(NAME check-rules COMMAND rudy-sentinel --check-rules --rules ${CMAKE_CURRENT_SOURCE_DIR}/rules/default.json)
set_tests_properties(check-rules PROPERTIES PASS_REGULAR_EXPRESSION "Rules valid: 14 rules")
if(RS_EVIDENCE)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME evidence-replay COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/evidence_replay.py $<TARGET_FILE:rudy-sentinel> ${CMAKE_CURRENT_SOURCE_DIR})
endif()
